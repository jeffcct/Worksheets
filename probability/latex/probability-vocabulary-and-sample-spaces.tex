\section*{Worked Example}
\noindent \textbf{Experiment:} Roll a fair six-sided die. \quad \textbf{Event:} rolling a number
greater than $4$.

\vspace{0.3em}
\begin{center}
\begin{tabular}{|T|D|}
\hline
Sample space & $\{1,2,3,4,5,6\}$ --- all possible outcomes. \\\hline
Outcome & $5$ --- just \emph{one} possible result (there are $6$ outcomes in total). \\\hline
Event & ``rolling a number greater than $4$'' --- the description of what we want. \\\hline
Event space & $\{5,6\}$ --- the outcomes that satisfy the event. \\\hline
Probability & $P(\text{event}) = \dfrac{2}{6} = \dfrac{1}{3}$ --- event space size $\div$ sample
space size. \\\hline
\end{tabular}
\end{center}

\section*{Calculating Probability}
\noindent\textbf{Reminder:} $P(\text{event}) = \dfrac{\text{event space size}}{\text{sample
space size}}$, written as a fraction in simplest form.

\vspace{0.3em}
\noindent\textbf{Worked example:} Spin a spinner numbered $1$--$4$. Event: landing on $1$.
Sample space $=\{1,2,3,4\}$, event space $=\{1\}$, so $P(\text{event}) = \dfrac{1}{4}$.

\vspace{0.5em}
\noindent\textbf{For each experiment, calculate the probability of the given event. Give your
answer as a fraction in simplest form.}

\begin{enumerate}[label=(\alph*)]
  \item Roll a die. Event: rolling a $5$.
  \anshere
  \item Toss a coin. Event: landing on heads.
  \anshere
  \item Spin a spinner numbered $1$--$8$. Event: spinning an even number.
  \anshere
  \item A bag has $4$ red and $6$ blue counters. Event: picking a blue counter.
  \anshere
  \item Draw a suit from a deck of cards. Event: drawing spades.
  \anshere
  \item Pick a letter from the word \textsc{probability}. Event: picking a vowel.
  \anshere
  \item Roll a die. Event: rolling a number less than $3$.
  \anshere
  \item A bag has $3$ yellow, $2$ green and $5$ purple marbles. Event: picking a green marble.
  \anshere
\end{enumerate}
