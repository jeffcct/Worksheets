\section*{Fluency}

\begin{multicols}{2}
\subsection*{Expand}
\noindent\textbf{Worked example:} Expand $2(x+5)$. Multiply everything inside the
brackets by $2$: $2(x+5) = 2\times x + 2\times 5 = 2x+10$.
\vspace{0.4em}
\begin{enumerate}[label=(\alph*)]
  \item $3(x+4)$
  \anshere
  \item $5(2x-3)$
  \anshere
  \item $-2(x+6)$
  \anshere
  \item $4(3x-5)$
  \anshere
  \item $-(x-7)$
  \anshere
  \item $x(x+3)$
  \anshere
  \item $2x(x-5)$
  \anshere
  \item $-3x(2x+1)$
  \anshere
\end{enumerate}

\subsection*{Factorise}
\noindent\textbf{Worked example:} Factorise $8x+20$. The HCF of $8x$ and $20$ is $4$, so
$8x+20 = 4(2x+5)$.
\vspace{0.4em}
\begin{enumerate}[label=(\alph*)]
  \item $4x+12$
  \anshere
  \item $6x-15$
  \anshere

  \anshere
  \item $3x^2-9x$
  \anshere
  \item $10x+25$
  \anshere
  \item $2x^2+8x$
  \anshere
  \item $9x-6$
  \anshere
  \item $9x^2-6x$
  \anshere
\end{enumerate}
\end{multicols}

\section*{Problem Solving}

\begin{multicols}{2}
\subsection*{Expand and simplify}
\noindent\textbf{Worked example:} Expand and simplify $2(x+4)+3(x-1)$. Expand each
bracket separately, then collect like terms:
\[2(x+4)+3(x-1) = 2x+8+3x-3 = 5x+5.\]
\noindent\textbf{Worked example:} Expand $(x+1)(x+6)$. Multiply each term in the first
bracket by each term in the second, then collect like terms:
\begin{align*}
(x+1)(x+6) &= x\times x + x\times 6 + 1\times x + 1\times 6\\
&= x^2+6x+x+6\\
&= x^2+7x+6.
\end{align*}
\vspace{0.2em}
\begin{enumerate}[label=(\alph*)]
  \item $3(x+2)+2(x-5)$
  \anshere
  \item $4(2x-1)-3(x+2)$
  \anshere
  \item $(x+3)(x+4)$
  \anshere
  \item $(2x-1)(x+5)$
  \anshere
  \item $(x-6)(x-2)$
  \anshere
\end{enumerate}

\subsection*{Factorise fully}
\noindent\textit{Find the highest common factor (HCF) of every term --- it can include
more than one letter, and a power of a letter.}

\vspace{0.3em}
\noindent\textbf{Worked example:} Factorise $12x^4y+18x^2y^3$. The HCF of $12$ and $18$
is $6$; the HCF of $x^4$ and $x^2$ is $x^2$; and $y$ is common to both terms. So the HCF
is $6x^2y$, giving
\[12x^4y+18x^2y^3 = 6x^2y(2x^2+3y^2).\]
\vspace{0.2em}
\begin{enumerate}[label=(\alph*)]
  \item $18x^5y^2+12x^7$
  \anshere
  \item $8a^3b^2-20a^2b^4$
  \anshere
  \item $15x^3y-9x^2y^2+6xy^3$
  \anshere
  \item $14p^4q^3+21p^2q^5$
  \anshere
  \item $24m^6n^2-16m^4n^5+8m^3n^3$
  \anshere
\end{enumerate}

\columnbreak

\subsection*{Solve}
\noindent\textit{Expand the brackets first, then solve for $x$.}

\vspace{0.3em}
\noindent\textbf{Worked example:} Solve $4(x-1)=16$. Expand the bracket:
$4x-4=16$. Add $4$ to both sides: $4x=20$. Divide both sides by $4$: $x=5$.
\vspace{0.2em}
\begin{enumerate}[label=(\alph*)]
  \item $3(x+5)=27$
  \anshere
  \item $2(3x-4)=5x+1$
  \anshere
  \item $5(x-2)-3(x+1)=7$
  \anshere
\end{enumerate}

\vspace{1em}
\subsection*{A step further}
\begin{enumerate}[label=(\alph*)]
  \item A rectangle has length $(x+5)$ and width $(x-2)$. Write an expression for its
    area, expanded and simplified.
  \anshere
  \item The length of a rectangle is $2x-3$ and its width is $x$. The area is
    $2x^2-3x$. Factorise this area expression to check it matches the length and
    width given.
  \anshere
\end{enumerate}
\end{multicols}

\section*{Enrichment}

\subsection*{Factorising Quadratics}
\noindent\textit{A quadratic $x^2+bx+c$ factorises as $(x+p)(x+q)$, where $p$ and $q$ are
two numbers that \emph{multiply} to give $c$ and \emph{add} to give $b$. A difference of
two squares, $a^2-b^2$, always factorises as $(a-b)(a+b)$.}

\vspace{0.4em}
\noindent\textbf{Worked example (sum and product):} Factorise $x^2+5x+6$. Two numbers that
multiply to $6$ and add to $5$ are $2$ and $3$, so $x^2+5x+6=(x+2)(x+3)$.

\vspace{0.4em}
\noindent\textbf{Worked example (difference of two squares):} Factorise $x^2-25$. Since
$x^2-25 = x^2-5^2$, this factorises as $x^2-25=(x-5)(x+5)$.

\vspace{0.4em}
\begin{enumerate}[label=(\alph*)]
  \item $x^2+7x+10$
  \anshere
  \item $x^2-3x-18$
  \anshere
  \item $x^2-16$
  \anshere
  \item $x^2+2x-15$
  \anshere
  \item $x^2-49$
  \anshere
\end{enumerate}

\subsection*{Inequalities}
\noindent\textit{Inequalities behave like equations, except multiplying or dividing by
a negative number reverses the inequality sign.}

\vspace{0.4em}
\noindent\textbf{Worked example:} Solve $2(x-3) < 10$.
\[2(x-3) < 10 \;\;\Rightarrow\;\; 2x-6 < 10 \;\;\Rightarrow\;\; 2x < 16 \;\;\Rightarrow\;\; x < 8.\]
(Expand the bracket, add $6$ to both sides, then divide both sides by $2$.)

\vspace{0.4em}
\noindent\textbf{Worked example (dividing by a negative):} Solve $-2x+5 > 1$.
\[-2x+5 > 1 \;\;\Rightarrow\;\; -2x > -4 \;\;\Rightarrow\;\; x < 2.\]
(Subtract $5$ from both sides, then divide both sides by $-2$ --- since we divided by a
negative number, the $>$ sign flips to $<$.)

\vspace{0.4em}
\begin{enumerate}[label=(\alph*)]
  \item Solve: \quad $3(x+2) > 15$
  \anshere
  \item Solve: \quad $2(x-5) \le 4(x+1)$
  \anshere
  \item Solve: \quad $-3(x+1) < 2(x-4)$
  \anshere
  \item Solve the double inequality: \quad $2 < 3(x-1) \le 11$
  \anshere
  \item A taxi charges a \$4 call-out fee plus \$3 per kilometre travelled. Write an
    inequality for the total cost of a trip of $k$ kilometres, then solve it to find
    how far you could travel and keep the cost under \$40.
  \anshere
\end{enumerate}
