\section*{Enrichment}

\subsection*{Factorising Quadratics}
\noindent\textit{A quadratic $x^2+bx+c$ (where $x^2$ has no coefficient) factorises as
$(x+p)(x+q)$, where $p$ and $q$ are two numbers that \emph{multiply} to give $c$ and
\emph{add} to give $b$. If the quadratic has a coefficient in front of the $x^2$, like
$ax^2+bx+c$, find two numbers that multiply to give $a\times c$ and add to give $b$, use
them to split the middle term into two terms, then factorise by grouping. A difference of
two squares, $a^2-b^2$, always factorises as $(a-b)(a+b)$.}

\vspace{0.4em}
\noindent\textbf{Worked example (no coefficient):} Factorise $x^2+5x+6$. Two numbers that
multiply to $6$ and add to $5$ are $2$ and $3$, so $x^2+5x+6=(x+2)(x+3)$.

\vspace{0.4em}
\noindent\textbf{Worked example (with a coefficient):} Factorise $2x^2+7x+3$. Here
$a\times c = 2\times 3=6$, and we need two numbers that multiply to $6$ and add to $7$:
these are $1$ and $6$. Split the middle term and group:
\[2x^2+7x+3 = 2x^2+x+6x+3 = x(2x+1)+3(2x+1) = (2x+1)(x+3).\]

\vspace{0.4em}
\begin{enumerate}[label=(\alph*)]
  \item $x^2+7x+10$
  \anshere
  \item $x^2-3x-18$
  \anshere
  \item $x^2-16$
  \anshere
  \item $4x^2-9$
  \anshere
  \item $2x^2+x-6$
  \anshere
\end{enumerate}

\subsection*{Inequalities}
\noindent\textit{Inequalities behave like equations, except multiplying or dividing by
a negative number reverses the inequality sign.}

\begin{enumerate}[label=(\alph*)]
  \item Solve: \quad $3(x+2) > 15$
  \anshere
  \item Solve: \quad $2(x-5) \le 4(x+1)$
  \anshere

  \anshere

  \anshere
\end{enumerate}
