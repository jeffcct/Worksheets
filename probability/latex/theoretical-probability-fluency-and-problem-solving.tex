\section*{Fluency}

\noindent\textbf{Theoretical probability:} ($\Omega$ = sample space, $E$ = event space)
\[
  P(E) = \frac{\text{number of items in } E}{\text{number of items in } \Omega}.
\]

\vspace{0.3em}
\noindent\textbf{Worked example:} A bag has $7$ red and $3$ blue counters. Find $P(\text{red})$.
\begin{quote}
\noindent $\Omega = \{7 \text{ red}, 3 \text{ blue}\}$ --- $10$ counters in total. \\
$E = \{7 \text{ red}\}$ --- $7$ counters are red. \\
$P(\text{red}) = \dfrac{7}{10}$.
\end{quote}

\vspace{0.3em}
\noindent\textbf{For each question: write down the sample space and the event space, then give
the probability as a fraction in simplest form.}

\begingroup\small
\setlist[enumerate,1]{leftmargin=1.4em, itemsep=0.3em, topsep=0.2em}
\begin{multicols}{2}
\begin{enumerate}[label=(\alph*)]
  \item Roll a die. Find $P(6)$.
  \answork
  \item Toss a coin. Find $P(\text{tails})$.
  \answork
  \item A bag has $10$ balls, $4$ of which are red. Find $P(\text{red})$.
  \answork
  \item A spinner has $5$ equal sections numbered $1$--$5$. Find $P(3)$.
  \answork
  \item A card is drawn from a deck of $52$. Find $P(\text{an Ace})$. (There are $4$ Aces.)
  \answork
  \item Roll a die. Find $P(\text{even number})$.
  \answork
  \item A bag has $3$ red, $2$ blue and $5$ green marbles. Find $P(\text{blue})$.
  \answork
  \item Toss a coin. Find $P(\text{heads})$.
  \answork
  \item A spinner has $4$ equal sections: red, blue, green, yellow. Find $P(\text{yellow})$.
  \answork
  \item Roll a die. Find $P(\text{a number greater than } 4)$.
  \answork
  \item A box has $8$ pens: $5$ black and $3$ blue. Find $P(\text{black})$.
  \answork
  \item A class has $20$ students, $12$ of whom are girls. Find $P(\text{picking a girl})$.
  \answork
\end{enumerate}

\section*{Problem Solving}

\noindent\textbf{Worked example:} A box of $24$ chocolates contains $6$ with a caramel filling.
Yui picks one chocolate at random. Find the probability it has a caramel filling.
\[
  P(\text{caramel filling}) = \frac{6}{24} = \frac{1}{4}.
\]

\vspace{0.5em}
\noindent\textbf{Give each answer as a fraction in simplest form.}

\begin{enumerate}[label=(\alph*)]
  \item A raffle sells $50$ tickets. Maya buys $4$ of them. One ticket is drawn at random. Find
    the probability that Maya wins.
  \anshere
  \item A classroom has $15$ boys and $10$ girls. The teacher picks one student at random to
    lead the assembly. Find the probability a girl is picked.
  \anshere
  \item A fruit bowl has $6$ apples, $4$ bananas and $2$ oranges. Sam picks one piece of fruit
    without looking. Find the probability Sam picks a banana.
  \anshere
  \item A board game spinner has $10$ equal sections: $6$ say ``move forward'' and $4$ say
    ``move back''. Find the probability of landing on ``move back''.
  \anshere
  \item A sports team of $22$ players includes $5$ goalkeepers. The coach randomly selects one
    player to interview. Find the probability the player is \emph{not} a goalkeeper.
  \anshere
  \item A jar of $36$ jellybeans contains $9$ that are lemon flavoured. Find the probability of
    picking a lemon jellybean.
  \anshere
  \item $8$ people enter a competition. Their names are put in a hat, and one is drawn at random
    to win first prize. Find the probability that Noah's name is drawn.
  \anshere
  \item A school has $400$ students, $25$ of whom are in the senior leadership team. One student
    is picked at random to represent the school at an event. Find the probability the student
    picked is on the leadership team.
  \anshere
\end{enumerate}

\vspace{0.3em}
\noindent\textbf{For the next two questions, listing out the event space will help.}

\begin{enumerate}[label=(\alph*), resume]
  \item A twenty-sided die (D$20$), numbered $1$ to $20$, is rolled. Find the probability of

  \answork
