\section{Simplifying Expressions}

\begin{multicols}{2}
\subsection*{Adding and subtracting like terms}
\begin{enumerate}
  \item $3x + 5x$ \ans
  \item $9y - 4y$ \ans
  \item $4a + 3a - 2a$ \ans
  \item $5x + 2y + 2x - y$ \ans
\end{enumerate}

\subsection*{Multiplying and dividing}
\begin{enumerate}[resume]
  \item $3 \times 4x$ \ans
  \item $(-2) \times 5a$ \ans
  \item $12x \div 4$ \ans
  \item $20y \div 5$ \ans
\end{enumerate}
\end{multicols}

\section{Solving Equations}

\begin{multicols}{2}
\subsection*{One-step equations}
\begin{enumerate}
  \item $x + 7 = 12$ \ans
  \item $x - 5 = 9$ \ans
  \item $4x = 24$ \ans
\end{enumerate}

\subsection*{Two-step equations}
\begin{enumerate}[resume]
  \item $2x + 3 = 11$ \ans
  \item $3x - 4 = 11$ \ans
  \item $\dfrac{x}{2} + 1 = 5$ \ans
\end{enumerate}

\subsection*{Multi-step equations}
\begin{enumerate}[resume]
  \item $2(x + 3) = 16$ \ans
  \item $3x + 2x - 1 = 14$ \ans
  \item $5x - 3 - x = 9$ \ans
\end{enumerate}

\subsection*{Variables on both sides}
\begin{enumerate}[resume]
  \item $3x + 2 = x + 10$ \ans
  \item $5x - 1 = 2x + 8$ \ans
  \item $4x + 3 = 2x + 11$ \ans
\end{enumerate}
\end{multicols}
